type credentialServer struct {
 mockControlServer
 template *pb.ActorTemplate
 actor *pb.Actor
 failTemplate bool
 poisonTemplate bool
 readyzTimeout int32
}
func (s *credentialServer) GetActorTemplate(_ context.Context,_ *pb.GetActorTemplateRequest)(*pb.ActorTemplate,error){
 if s.template==nil{return nil,status.Error(codes.NotFound,"missing")};return proto.Clone(s.template).(*pb.ActorTemplate),nil
}
func(s *credentialServer) CreateActorTemplate(_ context.Context,r *pb.CreateActorTemplateRequest)(*pb.ActorTemplate,error){
 if s.failTemplate{return nil,status.Error(codes.Internal,"test")}
 s.template=proto.Clone(r.ActorTemplate).(*pb.ActorTemplate)
 for _,container:=range s.template.Containers {
  if container.Readyz!=nil && container.Readyz.TimeoutSeconds==0 {container.Readyz.TimeoutSeconds=30}
  if s.readyzTimeout!=0 {container.Readyz.TimeoutSeconds=s.readyzTimeout}
 }
 if s.poisonTemplate{s.template.Containers[0].Env=append(s.template.Containers[0].Env,&pb.EnvVar{Name:"GEMINI_API_KEY",Value:"synthetic"})}
 return proto.Clone(s.template).(*pb.ActorTemplate),nil
}
func(s *credentialServer) CreateActor(_ context.Context,r *pb.CreateActorRequest)(*pb.Actor,error){
 if s.actor!=nil{return nil,status.Error(codes.AlreadyExists,"exists")}
 s.actor=proto.Clone(r.Actor).(*pb.Actor);s.actor.Status=&pb.ActorStatus{State:pb.ActorState_ACTOR_STATE_SUSPENDED}
 s.createdActors=append(s.createdActors,r.Actor.Metadata.Name)
 return proto.Clone(s.actor).(*pb.Actor),nil
}
func(s *credentialServer) GetActor(_ context.Context,_ *pb.GetActorRequest)(*pb.Actor,error){
 if s.actor==nil{return nil,status.Error(codes.NotFound,"missing")};return proto.Clone(s.actor).(*pb.Actor),nil
}
func credentialFixture(t *testing.T)(*controller.TaskReconciler,*credentialServer,*ax.Task,*int){
 t.Helper();t.Setenv("AX_DISABLE_CREDENTIAL_ATESPACES","ax-runtime");t.Setenv("GEMINI_API_KEY","synthetic-server-key")
 listener,err:=net.Listen("tcp","127.0.0.1:0");if err!=nil{t.Fatal(err)}
 service:=&credentialServer{};server:=grpc.NewServer();pb.RegisterControlServer(server,service);go server.Serve(listener)
 client,err:=substrate.NewClient(listener.Addr().String(),grpc.WithTransportCredentials(insecure.NewCredentials()));if err!=nil{t.Fatal(err)}
 t.Cleanup(func(){client.Close();server.Stop();listener.Close()})
 reconciler:=controller.NewTaskReconciler(client,"default-template","ax-system")
 calls:=new(int);reconciler.SecretResolver=func(context.Context,string,string,string)(string,error){*calls++;return "synthetic-secret-key",nil}
 task:=&ax.Task{Metadata:&ax.ObjectMeta{Name:"credential-test",Atespace:"ax-runtime"},Spec:&ax.TaskSpec{Image:"image@sha256:aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"}}
 return reconciler,service,task,calls
}
func TestCredentialFreeDoesNotReadOrInjectKeys(t *testing.T){
 r,s,task,calls:=credentialFixture(t);if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 if *calls!=0 || s.template==nil || len(s.template.Containers[0].Env)!=1 || s.template.Containers[0].Env[0].Name!="AX_TASK_YAML" {t.Fatal("credential source was used")}
}
func TestCredentialFreeAcceptsDefaultedReadiness(t *testing.T){
 r,s,task,_:=credentialFixture(t)
 if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 if s.template.Containers[0].Readyz.TimeoutSeconds!=30 || len(s.createdActors)!=1{t.Fatal("API defaulted readiness prevented actor creation")}
}
func TestCredentialFreeRejectsChangedReadiness(t *testing.T){
 r,s,task,_:=credentialFixture(t);s.readyzTimeout=31
 if _,err:=r.Reconcile(context.Background(),task);err==nil || len(s.createdActors)!=0{t.Fatal("unexpected readiness configuration accepted")}
}
func TestCredentialFreeFailedSavedTaskSuspendsWithoutResume(t *testing.T){
 r,s,task,_:=credentialFixture(t);task.Status=&ax.TaskStatus{Phase:"Failed"}
 store:=memory.NewStore();ctx:=context.Background()
 if err:=store.SaveTask(ctx,task);err!=nil{t.Fatal(err)}
 service:=server.NewServer(store,server.Options{Reconciler:r})
 observed,err:=service.GetTask(ctx,&ax.GetTaskRequest{Atespace:"ax-runtime",Name:task.Metadata.Name})
 if err!=nil || observed.Status.Phase!="Failed" || len(s.createdActors)!=0{t.Fatal("saved task was changed without an explicit operation")}
 stopped,err:=service.SuspendTask(ctx,&ax.SuspendTaskRequest{Atespace:"ax-runtime",Name:task.Metadata.Name})
 if err!=nil{t.Fatal(err)}
 if stopped.Status.Phase!="Suspended" || len(s.createdActors)!=1 || len(s.suspendedActors)!=1 || len(s.resumedActors)!=0 || len(s.revertedActors)!=0{t.Fatal("suspend restarted the saved task")}
}

 if _,err:=r.Reconcile(context.Background(),task);err==nil || len(s.createdActors)!=0{t.Fatal("task credential accepted")}
}

func TestCredentialFreeDoesNotRecoverCrashedSnapshot(t *testing.T){
 r,s,task,_:=credentialFixture(t);if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 s.actor.Status.State=pb.ActorState_ACTOR_STATE_CRASHED;task.Status.Phase="Running"
 if _,err:=r.Reconcile(context.Background(),task);err==nil || len(s.revertedActors)!=0 || len(s.resumedActors)!=0 {t.Fatal("snapshot recovered")}
}
func TestCredentialFreeCleanupRemainsPossible(t *testing.T){
 r,s,task,_:=credentialFixture(t);if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 s.failTemplate=true;s.actor.Status.State=pb.ActorState_ACTOR_STATE_RUNNING;task.Status.Phase="Suspended"
 if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 if len(s.suspendedActors)!=2{t.Fatal("cleanup missing")}
}
func TestCredentialPolicyKeepsLegacyAtespace(t *testing.T){
 r,s,task,calls:=credentialFixture(t);task.Metadata.Atespace="ax-demo"
 if _,err:=r.Reconcile(context.Background(),task);err!=nil{t.Fatal(err)}
 if *calls!=1{t.Fatal("legacy secret lookup changed")}
 found:=false;for _,e:=range s.template.Containers[0].Env{if e.Name=="GEMINI_API_KEY"{found=true}}
 if !found{t.Fatal("legacy injection removed")}
 r.SecretResolver=func(context.Context,string,string,string)(string,error){return "",errors.New("unavailable")}
}
